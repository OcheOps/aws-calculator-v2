# AWS Calculator — Plan 06: SvelteKit Frontend

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the actual product UX — service catalog browse, dynamic dimension form, live-priced estimate cart, save/share/export, magic-link sign-in, dark/light theming — all against the API from Plans 02–05. UI quality bar per spec: distinctive, tabular figures for money, sticky total on mobile, motion is subtle, dark mode from day one.

**Architecture:** SvelteKit 2 + Svelte 5 (runes) + Tailwind 4. A generated TypeScript client from a hand-maintained OpenAPI stub calls the Go API. SSR through the Node adapter; auth cookies flow via `event.cookies`. Client state (session, active currency, active org, current estimate cart) lives in Svelte stores hydrated on load. `DimensionForm.svelte` renders inputs dynamically from `service.dimensions[]` metadata.

**Spec:** `docs/superpowers/specs/2026-07-01-aws-calculator-design.md`
**Prior plans:** 01–05

- [ ] **Step 3: `client.ts`**

```ts
import { ApiError } from './errors';
import type { Envelope } from './types';

const API_BASE = import.meta.env.VITE_PUBLIC_API_URL || '';

type FetchImpl = typeof fetch;

export function makeApi(fetchImpl: FetchImpl = fetch) {
  async function request<T>(path: string, init: RequestInit = {}): Promise<T> {
    const res = await fetchImpl(API_BASE + '/api/v1' + path, {
      credentials: 'include',
      headers: { 'Content-Type': 'application/json', ...(init.headers ?? {}) },
      ...init,
    });
    const body = (await res.json().catch(() => ({}))) as Envelope<T>;
    if (!res.ok) {
      const err = body?.error;
      throw new ApiError(res.status, err?.code ?? 'error', err?.message ?? res.statusText);
    }
    return body.data;
  }
  return {
    providers: () => request<{code:string;name:string}[]>('/providers'),
    services: (q: { provider?: string; q?: string; cursor?: string; limit?: number }) => {
      const s = new URLSearchParams();
      Object.entries(q).forEach(([k,v]) => v !== undefined && s.set(k, String(v)));
      return request<any[]>('/services?' + s.toString());
    },
    service: (code: string, provider = 'aws') => request<{service: any; dimensions: any[]}>(`/services/${code}?provider=${provider}`),
    dimensionValues: (code: string, dim: string, provider = 'aws') =>
      request<any[]>(`/services/${code}/dimension-values?provider=${provider}&dim=${encodeURIComponent(dim)}`),
    regions: (provider = 'aws') => request<any[]>(`/regions?provider=${provider}`),
    currencies: () => request<string[]>('/currencies'),
    price: (in_: { service: string; region: string; term?: string; currency?: string; attributes: Record<string,string>; provider?: string }) => {
      const s = new URLSearchParams({ service: in_.service, region: in_.region, provider: in_.provider ?? 'aws' });
      if (in_.term) s.set('term', in_.term);
      if (in_.currency) s.set('currency', in_.currency);
      for (const [k,v] of Object.entries(in_.attributes)) s.set(`attr[${k}]`, v);
      return request<any>(`/price?${s.toString()}`);
    },
    // Auth
    requestLink: (email: string) => request<{accepted: boolean}>('/auth/request-link', { method: 'POST', body: JSON.stringify({ email }) }),
    verifyLink: (token: string, claim_tokens: string[] = []) => request<any>('/auth/verify', { method: 'POST', body: JSON.stringify({ token, claim_tokens }) }),
    logout: () => request<{ok:boolean}>('/auth/logout', { method: 'POST' }),
    me: () => request<any>('/auth/me'),
    // Estimates
    createEstimate: (body: { name: string; currency?: string; notes?: string }) => request<any>('/estimates', { method: 'POST', body: JSON.stringify(body) }),
    listEstimates: () => request<any[]>('/estimates'),
    getEstimate: (id: number, editToken?: string) => request<any>(`/estimates/${id}`, { headers: editToken ? { 'X-Edit-Token': editToken } : {} }),
    addItem: (id: number, body: any, editToken?: string) =>
      request<any>(`/estimates/${id}/items`, { method: 'POST', body: JSON.stringify(body), headers: editToken ? { 'X-Edit-Token': editToken } : {} }),
    reprice: (id: number, apply = false, editToken?: string) =>
      request<any>(`/estimates/${id}/reprice${apply ? '?apply=true' : ''}`, { method: 'POST', headers: editToken ? { 'X-Edit-Token': editToken } : {} }),
    share: (id: number, editToken?: string) =>
      request<{share_url:string; share_token:string}>(`/estimates/${id}/share`, { method: 'POST', headers: editToken ? { 'X-Edit-Token': editToken } : {} }),
    shared: (token: string) => request<any>(`/shared/${token}`),
  };
}

export const api = makeApi();
export type Api = ReturnType<typeof makeApi>;
```

- [ ] **Step 1: `hooks.server.ts`** — forwards the session cookie to the API and puts the resolved user into `event.locals`.

```ts
import type { Handle } from '@sveltejs/kit';

export const handle: Handle = async ({ event, resolve }) => {
  const session = event.cookies.get('session');
  if (session) {
    try {
      const res = await event.fetch((process.env.PUBLIC_API_URL || '') + '/api/v1/auth/me', {
        headers: { cookie: `session=${session}` }
      });
      if (res.ok) {
        const body = await res.json();
        event.locals.user = body.data;
      }
    } catch {}
  }
  return resolve(event);
};
```

Contract: `props = { service, dimensions, regions, onChange }`. Internal state: `values: Record<string,string>`, `region`, `dimensionValueCache`.

Behavior:
- Region select rendered first, always.
- For each `dimension` (skipping ones with `kind === 'region'`), render:
  - `enum` → Select loading options lazily via `api.dimensionValues(service.code, dim.key)` when region is set.
  - `numeric` → Input type=number.
  - `freeform` → Input.
- Emits `onChange({ region, term: 'OnDemand', attributes })` after debouncing 250ms.

Full code (~200 lines) — implementer writes out following the contract above with `$state`, `$effect`, and `$props()`.

Skeleton:

```svelte
<script lang="ts">
  import Select from './Select.svelte';
  import Input from './Input.svelte';
  import { debounce } from '$lib/utils/debounce';
  import { makeApi } from '$lib/api/client';
  const api = makeApi();

  let { service, dimensions, regions, onChange }: {
    service: any; dimensions: any[]; regions: {code:string}[]; onChange: (payload: any) => void;
  } = $props();

  let region = $state('');
  let values = $state<Record<string,string>>({});
  let dimValues = $state<Record<string, {value:string;label?:string}[]>>({});

  const emit = debounce(() => {
    if (!region) return;
    const dimsRequired = dimensions.filter(d => d.is_required && d.kind !== 'region').map(d => d.key);
    for (const k of dimsRequired) if (!values[k]) return;
    onChange({ region, term: 'OnDemand', attributes: values });
  }, 250);

  $effect(() => { void region; void JSON.stringify(values); emit(); });

  async function loadValues(dim: string) {
    if (dimValues[dim]) return;
    dimValues[dim] = await api.dimensionValues(service.code, dim);
  }
</script>

<div class="space-y-4">
  <Select bind:value={region} label="Region" options={regions.map(r => ({ value: r.code, label: r.code }))} />
  {#each dimensions.filter(d => d.kind !== 'region') as dim}
    {#if dim.kind === 'enum'}
      <div onfocusin={() => loadValues(dim.key)}>
        <Select bind:value={values[dim.key]} label={dim.label} options={dimValues[dim.key] ?? []} />
      </div>
    {:else if dim.kind === 'numeric'}
      <Input type="number" label={dim.label} bind:value={values[dim.key]} />
    {:else}
      <Input label={dim.label} bind:value={values[dim.key]} />
    {/if}
  {/each}
</div>
```

- [ ] **Step 3: `debounce.ts`**

```ts
export function debounce<F extends (...args: any[]) => void>(fn: F, ms: number): F {
  let t: any;
  return ((...args: any[]) => {
    clearTimeout(t);
    t = setTimeout(() => fn(...args), ms);
  }) as F;
}
```

```ts
import { redirect } from '@sveltejs/kit';

export const load = async ({ url, fetch, cookies }) => {
  const token = url.searchParams.get('token');
  if (!token) throw redirect(302, '/auth/request');
  const res = await fetch((process.env.PUBLIC_API_URL || '') + '/api/v1/auth/verify', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ token }),
  });
  const setCookie = res.headers.get('set-cookie');
  if (setCookie) {
    // Forward the session cookie set by the API.
    const match = /session=([^;]+)/.exec(setCookie);
    if (match) cookies.set('session', match[1], { path: '/', httpOnly: true, sameSite: 'lax', secure: process.env.API_ENV === 'production', maxAge: 30 * 24 * 3600 });
  }
  if (!res.ok) return { ok: false };
  throw redirect(302, '/');
};
```
