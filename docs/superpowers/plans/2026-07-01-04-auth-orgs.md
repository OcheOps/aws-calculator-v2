# AWS Calculator — Plan 04: Magic-Link Auth + Orgs

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Users sign in with email → click a magic link → receive a session cookie. A default organization is auto-created on first sign-in; the schema supports multi-user orgs but v1 does not expose invitations. Rate limits protect the request endpoint. Brevo (SMTP) delivers the emails.

**Architecture:** Passwordless. Server issues a 32-byte token; only its SHA-256 hash is stored (`magic_link_tokens`). On verify, the raw token is exchanged for a session (also stored as a hash) and a `session` cookie is set (HttpOnly, Secure, SameSite=Lax, 30-day sliding TTL). A `WithSession` middleware injects the current user into request context. In-memory token-bucket rate limiter per-IP and per-email guards `/auth/request-link`.

**Tech Stack:** Additions: `github.com/wneessen/go-mail` for Brevo SMTP; `golang.org/x/time/rate` (already present).

```go
package auth

import (
	"crypto/rand"
	"crypto/sha256"
	"encoding/base64"
	"encoding/hex"
)

func GenerateToken() (string, error) {
	var buf [32]byte
	if _, err := rand.Read(buf[:]); err != nil {
		return "", err
	}
	return base64.RawURLEncoding.EncodeToString(buf[:]), nil
}

func HashToken(raw string) string {
	sum := sha256.Sum256([]byte(raw))
	return hex.EncodeToString(sum[:])
}
```

```go
package auth

import (
	"context"
	"fmt"
	"regexp"
	"strings"
	"time"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgxpool"
)

type User struct {
	ID              int64
	Email           string
	EmailVerifiedAt *time.Time
}

type Org struct {
	ID   int64
	Name string
	Slug string
}

type Store struct{ pool *pgxpool.Pool }

func NewStore(p *pgxpool.Pool) *Store { return &Store{pool: p} }

func (s *Store) UpsertUser(ctx context.Context, email string) (*User, error) {
	email = strings.ToLower(strings.TrimSpace(email))
	var u User
	err := s.pool.QueryRow(ctx, `
		INSERT INTO users (email) VALUES ($1)
		ON CONFLICT (email) DO UPDATE SET email = EXCLUDED.email
		RETURNING id, email, email_verified_at`, email).Scan(&u.ID, &u.Email, &u.EmailVerifiedAt)
	return &u, err
}

func (s *Store) SaveMagicLink(ctx context.Context, userID int64, tokenHash string, expiresAt time.Time, ip, ua string) error {
	var ipVal interface{}
	if ip != "" {
		ipVal = ip
	}
	_, err := s.pool.Exec(ctx, `
		INSERT INTO magic_link_tokens (user_id, token_hash, expires_at, ip, ua)
		VALUES ($1, $2, $3, $4, $5)`, userID, tokenHash, expiresAt, ipVal, ua)
	return err
}

func (s *Store) ConsumeMagicLink(ctx context.Context, tokenHash string) (int64, error) {
	var userID int64
	err := s.pool.QueryRow(ctx, `
		UPDATE magic_link_tokens SET used_at = NOW()
		WHERE token_hash = $1 AND used_at IS NULL AND expires_at > NOW()
		RETURNING user_id`, tokenHash).Scan(&userID)
	if err != nil {
		return 0, fmt.Errorf("invalid or expired token")
	}
	// Mark user verified.
	_, _ = s.pool.Exec(ctx, `UPDATE users SET email_verified_at = COALESCE(email_verified_at, NOW()) WHERE id = $1`, userID)
	return userID, nil
}

func (s *Store) SaveSession(ctx context.Context, userID int64, tokenHash string, expiresAt time.Time) error {
	_, err := s.pool.Exec(ctx,
		`INSERT INTO sessions (user_id, token_hash, expires_at) VALUES ($1, $2, $3)`,
		userID, tokenHash, expiresAt)
	return err
}

func (s *Store) LookupSession(ctx context.Context, tokenHash string) (*User, error) {
	var u User
	var expiresAt time.Time
	err := s.pool.QueryRow(ctx, `
		SELECT u.id, u.email, u.email_verified_at, s.expires_at FROM sessions s
		JOIN users u ON u.id = s.user_id
		WHERE s.token_hash = $1 AND u.disabled_at IS NULL`, tokenHash).Scan(&u.ID, &u.Email, &u.EmailVerifiedAt, &expiresAt)
	if err != nil {
		return nil, fmt.Errorf("session not found")
	}
	if time.Now().After(expiresAt) {
		return nil, fmt.Errorf("session expired")
	}
	_, _ = s.pool.Exec(ctx, `UPDATE sessions SET last_seen_at = NOW() WHERE token_hash = $1`, tokenHash)
	return &u, nil
}

func (s *Store) DeleteSession(ctx context.Context, tokenHash string) error {
	_, err := s.pool.Exec(ctx, `DELETE FROM sessions WHERE token_hash = $1`, tokenHash)
	return err
}

var slugSanitizer = regexp.MustCompile(`[^a-z0-9-]+`)

func slugify(local string) string {
	local = strings.ToLower(local)
	local = slugSanitizer.ReplaceAllString(local, "-")
	return strings.Trim(local, "-")
}

func (s *Store) EnsureDefaultOrg(ctx context.Context, u *User) (*Org, error) {
	// If user already has a membership, return the first org.
	var existing Org
	err := s.pool.QueryRow(ctx, `
		SELECT o.id, o.name, o.slug FROM memberships m
		JOIN organizations o ON o.id = m.org_id
		WHERE m.user_id = $1 ORDER BY m.created_at ASC LIMIT 1`, u.ID).Scan(&existing.ID, &existing.Name, &existing.Slug)
	if err == nil {
		return &existing, nil
	}
	// Create new org.
	local := u.Email
	if i := strings.Index(u.Email, "@"); i > 0 {
		local = u.Email[:i]
	}
	base := slugify(local)
	if base == "" {
		base = "user"
	}
	name := fmt.Sprintf("%s's workspace", local)
	slug := base
	tx, err := s.pool.BeginTx(ctx, pgx.TxOptions{})
	if err != nil {
		return nil, err
	}
	defer tx.Rollback(ctx)
	for i := 0; i < 10; i++ {
		var org Org
		err := tx.QueryRow(ctx, `INSERT INTO organizations (name, slug) VALUES ($1, $2) RETURNING id, name, slug`, name, slug).Scan(&org.ID, &org.Name, &org.Slug)
		if err == nil {
			if _, err := tx.Exec(ctx, `INSERT INTO memberships (org_id, user_id, role) VALUES ($1, $2, 'owner')`, org.ID, u.ID); err != nil {
				return nil, err
			}
			if err := tx.Commit(ctx); err != nil {
				return nil, err
			}
			return &org, nil
		}
		slug = fmt.Sprintf("%s-%d", base, i+1)
	}
	return nil, fmt.Errorf("could not generate unique slug")
}

func (s *Store) GetOrgsForUser(ctx context.Context, userID int64) ([]Org, error) {
	rows, err := s.pool.Query(ctx, `SELECT o.id, o.name, o.slug FROM memberships m JOIN organizations o ON o.id = m.org_id WHERE m.user_id = $1 ORDER BY o.name`, userID)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var out []Org
	for rows.Next() {
		var o Org
		if err := rows.Scan(&o.ID, &o.Name, &o.Slug); err != nil {
			return nil, err
		}
		out = append(out, o)
	}
	return out, rows.Err()
}
```

```go
package auth

import (
	"sync"
	"time"

	"golang.org/x/time/rate"
)

type Limiter struct {
	mu       sync.Mutex
	burst    int
	interval time.Duration
	buckets  map[string]*rate.Limiter
	lastSeen map[string]time.Time
}

// NewLimiter allows `burst` events per `interval` (e.g., 5 / 1 hour).
func NewLimiter(burst int, interval time.Duration) *Limiter {
	return &Limiter{
		burst:    burst,
		interval: interval,
		buckets:  map[string]*rate.Limiter{},
		lastSeen: map[string]time.Time{},
	}
}

func (l *Limiter) Allow(key string) bool {
	l.mu.Lock()
	defer l.mu.Unlock()
	b, ok := l.buckets[key]
	if !ok {
		b = rate.NewLimiter(rate.Every(l.interval/time.Duration(l.burst)), l.burst)
		l.buckets[key] = b
	}
	l.lastSeen[key] = time.Now()
	return b.Allow()
}
```

```go
package mail

import (
	"crypto/tls"
	"fmt"

	"github.com/wneessen/go-mail"
)

type Mailer interface {
	Send(to, subject, body string) error
}

// Brevo SMTP implementation.
type SMTPMailer struct {
	host, user, pass, from string
	port                   int
}

func NewSMTPMailer(host string, port int, user, pass, from string) *SMTPMailer {
	return &SMTPMailer{host: host, port: port, user: user, pass: pass, from: from}
}

func (s *SMTPMailer) Send(to, subject, body string) error {
	m := mail.NewMsg()
	if err := m.From(s.from); err != nil {
		return err
	}
	if err := m.To(to); err != nil {
		return err
	}
	m.Subject(subject)
	m.SetBodyString(mail.TypeTextHTML, body)

	c, err := mail.NewClient(s.host,
		mail.WithPort(s.port),
		mail.WithSMTPAuth(mail.SMTPAuthPlain),
		mail.WithUsername(s.user),
		mail.WithPassword(s.pass),
		mail.WithTLSConfig(&tls.Config{ServerName: s.host}),
		mail.WithTLSPolicy(mail.TLSMandatory),
	)
	if err != nil {
		return fmt.Errorf("smtp client: %w", err)
	}
	return c.DialAndSend(m)
}

// Fake for tests.
type FakeMailer struct {
	Messages []Message
}

type Message struct {
	To, Subject, Body string
}

func NewFake() *FakeMailer { return &FakeMailer{} }

func (f *FakeMailer) Send(to, subject, body string) error {
	f.Messages = append(f.Messages, Message{to, subject, body})
	return nil
}
```

```go
package auth

import (
	"context"
	"fmt"
	"time"

	"github.com/OcheOps/aws-calculator/api/internal/mail"
)

type Service struct {
	store        *Store
	mailer       mail.Mailer
	byIP         *Limiter
	byEmail      *Limiter
	baseURL      string
	linkTTL      time.Duration
	sessionTTL   time.Duration
}

func NewService(store *Store, m mail.Mailer, byIP, byEmail *Limiter, baseURL string) *Service {
	return &Service{
		store: store, mailer: m,
		byIP: byIP, byEmail: byEmail,
		baseURL: baseURL,
		linkTTL: 15 * time.Minute,
		sessionTTL: 30 * 24 * time.Hour,
	}
}

func (s *Service) RequestLink(ctx context.Context, email, ip, ua string) error {
	if !s.byIP.Allow(ip) || !s.byEmail.Allow(email) {
		return fmt.Errorf("rate limited")
	}
	user, err := s.store.UpsertUser(ctx, email)
	if err != nil {
		return err
	}
	raw, err := GenerateToken()
	if err != nil {
		return err
	}
	if err := s.store.SaveMagicLink(ctx, user.ID, HashToken(raw), time.Now().Add(s.linkTTL), ip, ua); err != nil {
		return err
	}
	link := fmt.Sprintf("%s/auth/verify?token=%s", s.baseURL, raw)
	body := fmt.Sprintf(`<p>Click to sign in:</p><p><a href="%s">%s</a></p><p>This link expires in 15 minutes.</p>`, link, link)
	return s.mailer.Send(user.Email, "Your AWS Calculator sign-in link", body)
}

// Verify consumes a magic-link token, issues a session, and returns the raw
// session token (caller sets the cookie), the session expiry, and the user.
func (s *Service) Verify(ctx context.Context, rawToken string, claimTokens []string) (*User, string, time.Time, error) {
	uid, err := s.store.ConsumeMagicLink(ctx, HashToken(rawToken))
	if err != nil {
		return nil, "", time.Time{}, err
	}
	user, err := s.getUser(ctx, uid)
	if err != nil {
		return nil, "", time.Time{}, err
	}
	if _, err := s.store.EnsureDefaultOrg(ctx, user); err != nil {
		return nil, "", time.Time{}, err
	}
	sessRaw, err := GenerateToken()
	if err != nil {
		return nil, "", time.Time{}, err
	}
	exp := time.Now().Add(s.sessionTTL)
	if err := s.store.SaveSession(ctx, uid, HashToken(sessRaw), exp); err != nil {
		return nil, "", time.Time{}, err
	}
	return user, sessRaw, exp, nil
}

func (s *Service) Logout(ctx context.Context, sessionRaw string) error {
	return s.store.DeleteSession(ctx, HashToken(sessionRaw))
}

func (s *Service) Session(ctx context.Context, sessionRaw string) (*User, error) {
	return s.store.LookupSession(ctx, HashToken(sessionRaw))
}

func (s *Service) getUser(ctx context.Context, id int64) (*User, error) {
	var u User
	err := s.store.pool.QueryRow(ctx, `SELECT id, email, email_verified_at FROM users WHERE id = $1`, id).Scan(&u.ID, &u.Email, &u.EmailVerifiedAt)
	if err != nil {
		return nil, err
	}
	return &u, nil
}
```

```go
package httpapi

import (
	"context"
	"net/http"

	"github.com/OcheOps/aws-calculator/api/internal/auth"
)

type ctxKey int

const ctxKeyUser ctxKey = iota

func WithSession(svc *auth.Service) func(http.Handler) http.Handler {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			cookie, err := r.Cookie("session")
			if err == nil && cookie.Value != "" {
				if user, err := svc.Session(r.Context(), cookie.Value); err == nil {
					ctx := context.WithValue(r.Context(), ctxKeyUser, user)
					r = r.WithContext(ctx)
				}
			}
			next.ServeHTTP(w, r)
		})
	}
}

func UserFrom(ctx context.Context) (*auth.User, bool) {
	u, ok := ctx.Value(ctxKeyUser).(*auth.User)
	return u, ok
}
```

```go
package httpapi

import (
	"encoding/json"
	"net/http"
	"time"

	"github.com/OcheOps/aws-calculator/api/internal/auth"
	"github.com/go-chi/chi/v5"
)

type AuthOptions struct {
	SecureCookie bool // true in production
	CookieDomain string
}

func MountAuth(r chi.Router, svc *auth.Service, opts AuthOptions) {
	r.Post("/auth/request-link", func(w http.ResponseWriter, r *http.Request) {
		var body struct {
			Email string `json:"email"`
		}
		if err := json.NewDecoder(r.Body).Decode(&body); err != nil || body.Email == "" {
			WriteError(w, http.StatusBadRequest, "bad_request", "email is required")
			return
		}
		_ = svc.RequestLink(r.Context(), body.Email, clientIP(r), r.UserAgent())
		// Always 202 to avoid enumeration.
		WriteData(w, http.StatusAccepted, map[string]bool{"accepted": true}, nil)
	})

	r.Post("/auth/verify", func(w http.ResponseWriter, r *http.Request) {
		var body struct {
			Token        string   `json:"token"`
			ClaimTokens  []string `json:"claim_tokens"`
		}
		if err := json.NewDecoder(r.Body).Decode(&body); err != nil || body.Token == "" {
			WriteError(w, http.StatusBadRequest, "bad_request", "token is required")
			return
		}
		user, sessionRaw, exp, err := svc.Verify(r.Context(), body.Token, body.ClaimTokens)
		if err != nil {
			WriteError(w, http.StatusUnauthorized, "unauthorized", err.Error())
			return
		}
		http.SetCookie(w, &http.Cookie{
			Name:     "session",
			Value:    sessionRaw,
			Path:     "/",
			Domain:   opts.CookieDomain,
			Expires:  exp,
			HttpOnly: true,
			Secure:   opts.SecureCookie,
			SameSite: http.SameSiteLaxMode,
		})
		WriteData(w, http.StatusOK, map[string]any{
			"user":       user,
			"expires_at": exp.UTC().Format(time.RFC3339),
		}, nil)
	})

	r.Post("/auth/logout", func(w http.ResponseWriter, r *http.Request) {
		if c, err := r.Cookie("session"); err == nil {
			_ = svc.Logout(r.Context(), c.Value)
		}
		http.SetCookie(w, &http.Cookie{
			Name: "session", Value: "", Path: "/", MaxAge: -1,
			HttpOnly: true, Secure: opts.SecureCookie, SameSite: http.SameSiteLaxMode,
		})
		WriteData(w, http.StatusOK, map[string]bool{"ok": true}, nil)
	})

	r.Get("/auth/me", func(w http.ResponseWriter, r *http.Request) {
		u, ok := UserFrom(r.Context())
		if !ok {
			WriteError(w, http.StatusUnauthorized, "unauthorized", "not signed in")
			return
		}
		WriteData(w, http.StatusOK, u, nil)
	})
}

func clientIP(r *http.Request) string {
	if v := r.Header.Get("X-Forwarded-For"); v != "" {
		return v
	}
	return r.RemoteAddr
}
```
