```go
package fx

import (
	"context"
	"fmt"

	"github.com/jackc/pgx/v5/pgxpool"
	"github.com/shopspring/decimal"
)

type Store struct{ pool *pgxpool.Pool }

func NewStore(pool *pgxpool.Pool) *Store { return &Store{pool: pool} }

func (s *Store) Upsert(ctx context.Context, rates []Rate) error {
	for _, r := range rates {
		if _, err := s.pool.Exec(ctx,
			`INSERT INTO fx_rates (base, quote, rate) VALUES ($1, $2, $3::numeric)`,
			r.Base, r.Quote, r.Rate.String()); err != nil {
			return fmt.Errorf("insert fx %s/%s: %w", r.Base, r.Quote, err)
		}
	}
	return nil
}

func (s *Store) Latest(ctx context.Context, base, quote string) (Rate, error) {
	var rateStr string
	var r Rate
	err := s.pool.QueryRow(ctx,
		`SELECT rate::text, fetched_at FROM fx_rates WHERE base=$1 AND quote=$2 ORDER BY fetched_at DESC LIMIT 1`,
		base, quote).Scan(&rateStr, &r.FetchedAt)
	if err != nil {
		return r, fmt.Errorf("no rate for %s/%s: %w", base, quote, err)
	}
	dec, err := decimal.NewFromString(rateStr)
	if err != nil {
		return r, fmt.Errorf("parse rate: %w", err)
	}
	r.Base, r.Quote, r.Rate = base, quote, dec
	return r, nil
}
```

```go
package fx

import (
	"context"
	"encoding/json"
	"fmt"
	"net/http"
	"strings"
	"time"

	"github.com/shopspring/decimal"
)

type Fetcher interface {
	Fetch(ctx context.Context, base string, quotes []string) ([]Rate, error)
}

type HTTPClient struct {
	baseURL string
	http    *http.Client
}

func NewHTTPClient(baseURL string) *HTTPClient {
	return &HTTPClient{baseURL: baseURL, http: &http.Client{Timeout: 10 * time.Second}}
}

type fxResponse struct {
	Base  string             `json:"base"`
	Rates map[string]float64 `json:"rates"`
}

func (c *HTTPClient) Fetch(ctx context.Context, base string, quotes []string) ([]Rate, error) {
	url := fmt.Sprintf("%s/latest?base=%s&symbols=%s", c.baseURL, base, strings.Join(quotes, ","))
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, nil)
	if err != nil {
		return nil, err
	}
	resp, err := c.http.Do(req)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()
	if resp.StatusCode >= 400 {
		return nil, fmt.Errorf("fx fetch %d", resp.StatusCode)
	}
	var body fxResponse
	if err := json.NewDecoder(resp.Body).Decode(&body); err != nil {
		return nil, err
	}
	now := time.Now().UTC()
	rates := make([]Rate, 0, len(body.Rates))
	for quote, v := range body.Rates {
		rates = append(rates, Rate{Base: base, Quote: quote, Rate: decimal.NewFromFloat(v), FetchedAt: now})
	}
	return rates, nil
}
```

- [ ] **Step 3: Implement `store.go`**

Algorithm: build the attributes JSON, query with `attributes @> $1`. If no row, load required dimension keys, drop optional attrs one at a time in reverse `sort_order`, retry.

```go
package pricing

import (
	"context"
	"encoding/json"
	"fmt"
	"sort"

	"github.com/jackc/pgx/v5/pgxpool"
	"github.com/shopspring/decimal"
)

type Store struct{ pool *pgxpool.Pool }

func NewStore(pool *pgxpool.Pool) *Store { return &Store{pool: pool} }

type dim struct {
	Key        string
	IsRequired bool
	SortOrder  int
}

func (s *Store) Find(ctx context.Context, q LookupQuery) (*LookupResult, error) {
	dims, err := s.dimensions(ctx, q.ServiceID)
	if err != nil {
		return nil, err
	}
	if err := s.checkRequired(q.Attributes, dims); err != nil {
		return nil, err
	}

	if r, err := s.queryOne(ctx, q, q.Attributes); err == nil {
		r.MatchQuality = "exact"
		return r, nil
	}

	// Fallback: drop optional attrs from lowest priority (highest sort_order) first.
	optional := make([]string, 0)
	for _, d := range dims {
		if !d.IsRequired {
			if _, present := q.Attributes[d.Key]; present {
				optional = append(optional, d.Key)
			}
		}
	}
	sort.SliceStable(optional, func(i, j int) bool {
		return dimSortOrder(dims, optional[i]) > dimSortOrder(dims, optional[j])
	})

	dropped := make([]string, 0, len(optional))
	attrs := copyMap(q.Attributes)
	for _, k := range optional {
		delete(attrs, k)
		dropped = append(dropped, k)
		if r, err := s.queryOne(ctx, q, attrs); err == nil {
			r.MatchQuality = "partial"
			r.DroppedAttrs = dropped
			return r, nil
		}
	}
	return nil, fmt.Errorf("no matching sku")
}

func (s *Store) queryOne(ctx context.Context, q LookupQuery, attrs map[string]string) (*LookupResult, error) {
	attrJSON, err := json.Marshal(attrs)
	if err != nil {
		return nil, err
	}
	var (
		sku, unit, currency, term string
		priceStr                  string
	)
	err = s.pool.QueryRow(ctx, `
		SELECT sku, unit, currency, price_per_unit::text, term_type
		FROM sku_prices
		WHERE service_id=$1 AND region_code=$2 AND term_type=$3 AND attributes @> $4::jsonb
		ORDER BY price_per_unit ASC
		LIMIT 1`, q.ServiceID, q.Region, q.TermType, attrJSON).Scan(&sku, &unit, &currency, &priceStr, &term)
	if err != nil {
		return nil, err
	}
	price, err := decimal.NewFromString(priceStr)
	if err != nil {
		return nil, err
	}
	return &LookupResult{SKU: sku, Unit: unit, Currency: currency, PricePerUnit: price, TermType: term}, nil
}

func (s *Store) dimensions(ctx context.Context, serviceID int64) ([]dim, error) {
	rows, err := s.pool.Query(ctx, `SELECT key, is_required, sort_order FROM service_dimensions WHERE service_id=$1`, serviceID)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var out []dim
	for rows.Next() {
		var d dim
		if err := rows.Scan(&d.Key, &d.IsRequired, &d.SortOrder); err != nil {
			return nil, err
		}
		out = append(out, d)
	}
	return out, rows.Err()
}

func (s *Store) checkRequired(attrs map[string]string, dims []dim) error {
	for _, d := range dims {
		if d.IsRequired {
			if _, ok := attrs[d.Key]; !ok {
				return fmt.Errorf("required attribute %q missing", d.Key)
			}
		}
	}
	return nil
}

func dimSortOrder(dims []dim, key string) int {
	for _, d := range dims {
		if d.Key == key {
			return d.SortOrder
		}
	}
	return 0
}

func copyMap(m map[string]string) map[string]string {
	out := make(map[string]string, len(m))
	for k, v := range m {
		out[k] = v
	}
	return out
}
```

```go
package pricing

import (
	"time"

	"github.com/dgraph-io/ristretto/v2"
)

type Cache struct {
	c   *ristretto.Cache[string, *LookupResult]
	ttl time.Duration
}

func NewCache(maxCost int64, ttl time.Duration) (*Cache, error) {
	c, err := ristretto.NewCache(&ristretto.Config[string, *LookupResult]{
		NumCounters: maxCost * 10,
		MaxCost:     maxCost,
		BufferItems: 64,
	})
	if err != nil {
		return nil, err
	}
	return &Cache{c: c, ttl: ttl}, nil
}

func (c *Cache) Get(key string) (*LookupResult, bool) {
	return c.c.Get(key)
}

func (c *Cache) Set(key string, val *LookupResult) {
	c.c.SetWithTTL(key, val, 1, c.ttl)
}

func CacheKey(serviceCode, region, term string, attrs map[string]string) string {
	// stable order
	keys := make([]string, 0, len(attrs))
	for k := range attrs {
		keys = append(keys, k)
	}
	sortStrings(keys)
	buf := serviceCode + "|" + region + "|" + term
	for _, k := range keys {
		buf += "|" + k + "=" + attrs[k]
	}
	return buf
}

func sortStrings(s []string) {
	// tiny helper to avoid pulling in sort just to inline call
	for i := 1; i < len(s); i++ {
		for j := i; j > 0 && s[j-1] > s[j]; j-- {

		}
	}
```

```go
package pricing

import (
	"context"
	"fmt"
	"time"

	"github.com/OcheOps/aws-calculator/api/internal/catalog"
	"github.com/OcheOps/aws-calculator/api/internal/fx"
	"github.com/shopspring/decimal"
)

type LookupInput struct {
	Provider    string
	ServiceCode string
	Region      string
	TermType    string
	Attributes  map[string]string
	Currency    string
}

type PriceResponse struct {
	*LookupResult
	Currency string          `json:"currency"`
	FXRate   decimal.Decimal `json:"fx_rate,omitempty"`
	FXAsOf   *time.Time      `json:"fx_as_of,omitempty"`
}

type Lookup struct {
	store   *Store
	fx      *fx.Store
	cache   *Cache
	catalog *catalog.Store
}

func NewLookup(s *Store, f *fx.Store, c *Cache, cat *catalog.Store) *Lookup {
	return &Lookup{store: s, fx: f, cache: c, catalog: cat}
}

func (l *Lookup) Price(ctx context.Context, in LookupInput) (*PriceResponse, error) {
	if in.TermType == "" {
		in.TermType = "OnDemand"
	}
	if in.Currency == "" {
		in.Currency = "USD"
	}
	svc, err := l.catalog.GetServiceByCode(ctx, in.Provider, in.ServiceCode)
	if err != nil {
		return nil, fmt.Errorf("service not found: %w", err)
	}

	key := CacheKey(in.ServiceCode, in.Region, in.TermType, in.Attributes) + "|" + in.Currency
	if cached, ok := l.cache.Get(key); ok {
		return l.wrap(ctx, cached, in.Currency)
	}

	res, err := l.store.Find(ctx, LookupQuery{
		ServiceID: svc.ID, ServiceCode: in.ServiceCode, Region: in.Region,
		TermType: in.TermType, Attributes: in.Attributes,
	})
	if err != nil {
		return nil, err
	}
	l.cache.Set(key, res)
	return l.wrap(ctx, res, in.Currency)
}

func (l *Lookup) wrap(ctx context.Context, res *LookupResult, currency string) (*PriceResponse, error) {
	resp := &PriceResponse{LookupResult: res, Currency: currency}
	if currency == res.Currency {
		return resp, nil
	}
	rate, err := l.fx.Latest(ctx, res.Currency, currency)
	if err != nil {
		return nil, fmt.Errorf("no fx rate for %s/%s", res.Currency, currency)
	}
	converted := res.PricePerUnit.Mul(rate.Rate)
	// Copy result to avoid mutating cached value.
	newRes := *res
	newRes.PricePerUnit = converted
	newRes.Currency = currency
	resp.LookupResult = &newRes
	resp.FXRate = rate.Rate
	fetched := rate.FetchedAt
	resp.FXAsOf = &fetched
	return resp, nil
}
```

- [ ] **Step 2: Implement `price.go`**

Query parsing: attributes come in as `attr[<key>]=<value>` (bracket syntax). Parse with `net/url` — Go's `url.Values` already handles this because `[` and `]` are literal characters in the key.

```go
package httpapi

import (
	"net/http"
	"strings"

	"github.com/OcheOps/aws-calculator/api/internal/pricing"
	"github.com/go-chi/chi/v5"
)

func MountPricing(r chi.Router, lookup *pricing.Lookup) {
	r.Get("/price", func(w http.ResponseWriter, r *http.Request) {
		q := r.URL.Query()
		attrs := map[string]string{}
		for k, vs := range q {
			if strings.HasPrefix(k, "attr[") && strings.HasSuffix(k, "]") && len(vs) > 0 {
				name := k[len("attr[") : len(k)-1]
				attrs[name] = vs[0]
			}
		}
		in := pricing.LookupInput{
			Provider:    stringOr(q.Get("provider"), "aws"),
			ServiceCode: q.Get("service"),
			Region:      q.Get("region"),
			TermType:    stringOr(q.Get("term"), "OnDemand"),
			Attributes:  attrs,
			Currency:    stringOr(q.Get("currency"), "USD"),
		}
		if in.ServiceCode == "" || in.Region == "" {
			WriteError(w, http.StatusBadRequest, "bad_request", "service and region are required")
			return
		}
		res, err := lookup.Price(r.Context(), in)
		if err != nil {
			WriteError(w, http.StatusNotFound, "not_found", err.Error())
			return
		}
		WriteData(w, http.StatusOK, res, nil)
	})
}

func stringOr(v, def string) string {
	if v == "" {
		return def
	}
	return v
}
```
